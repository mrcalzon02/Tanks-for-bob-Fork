-- Space Age / Quality guardrails for the corporate waste economy.
-- This file is intentionally a compatibility patch, not a content add-on.

if not mods["space-age"] then return end

local function patch_recipe(name, fields)
  local recipe = data.raw.recipe[name]
  if not recipe then return end

  for key, value in pairs(fields) do
    recipe[key] = value
  end
end
